# Punch Club 3D - alternativa ao Makefile (Visual Studio, CLion, Code::Blocks via CMake)
# Windows + vcpkg:  vcpkg install freeglut  e  cmake -B build -DCMAKE_TOOLCHAIN_FILE=<vcpkg>/scripts/buildsystems/vcpkg.cmake
cmake_minimum_required(VERSION 3.10)
project(PunchClub3D CXX)
set(CMAKE_CXX_STANDARD 17)
find_package(OpenGL REQUIRED)
find_package(GLUT REQUIRED)
file(GLOB SRC src/*.cpp)
add_executable(punchclub3d ${SRC})
target_include_directories(punchclub3d PRIVATE third_party ${GLUT_INCLUDE_DIR})
target_link_libraries(punchclub3d PRIVATE GLUT::GLUT OpenGL::GL OpenGL::GLU)
if(UNIX AND NOT APPLE)
  target_link_libraries(punchclub3d PRIVATE pthread dl m)
endif()
